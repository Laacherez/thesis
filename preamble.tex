% ============================================================

\newif\ifshowchaptertabs
\showchaptertabstrue

\newcommand{\enablechaptertabs}{\global\showchaptertabstrue}
\newcommand{\disablechaptertabs}{\global\showchaptertabsfalse}

\newcommand{\chaptertab}{%
  \AddToShipoutPictureBG{%
    \begin{tikzpicture}[remember picture, overlay]
      \ifshowchaptertabs
        \ifnum\value{chapter}>0
          \ifodd\value{page}
            % Right-hand pages: tab on the right
            \node[
              fill=black,
              text=white,
              font=\sffamily\bfseries\small,
              minimum width=0.8cm,
              minimum height=1.8cm,
              anchor=east
            ] at ([yshift={-2.5cm-\value{chapter}*1.9cm}]current page.north east)
            {\thechapter};
          \else
            % Left-hand pages: tab on the left
            \node[
              fill=black,
              text=white,
              font=\sffamily\bfseries\small,
              minimum width=0.8cm,
              minimum height=1.8cm,
              anchor=west
            ] at ([yshift={-2.5cm-\value{chapter}*1.9cm}]current page.north west)
            {\thechapter};
          \fi
        \fi
      \fi
    \end{tikzpicture}%
  }%
}

\AtBeginDocument{\chaptertab}

% ============================================================
% Publication appendix helpers
% ============================================================

\newcommand{\currentpublicationtitle}{}

\newcommand{\publicationsection}[1]{%
  \clearpage
  \refstepcounter{section}%
  \renewcommand{\currentpublicationtitle}{#1}%
  \phantomsection
  \addcontentsline{toc}{section}{\protect\numberline{\thesection}#1}%
  \markright{\thesection\quad #1}%
}

\newcommand{\includepublicationpdf}[2][]{%
  \includepdf[
    pages=-,
    scale=0.85,
    pagecommand={%
      \thispagestyle{fancy}%
      \markright{\thesection\quad \currentpublicationtitle}%
    },
    #1
  ]{#2}%
}
